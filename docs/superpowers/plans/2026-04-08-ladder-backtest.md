"""
Chunk 22: LADDER BACKTEST — config-driven simulation of sa_spins-based
bet-ladder strategies against the 28k+ spin dataset, with per-account
comparison vs actual historical play.

For each strategy:
  - Walk each account's spins, computing a session-local sa_spins counter
    (fresh at each new CSV, resets on each in-session accumulation triple).
  - Look up bet multiplier from the ladder.
  - Compute capture rate / efficiency / waste at the ladder's top tier.
  - Compute the same metrics on actual historical bet_multiplier data
    using the same top-tier threshold (apples-to-apples comparison).
  - Emit a per-account report line.

Add new strategies by pasting a new entry into STRATEGIES and re-running.
"""
import pickle
from pathlib import Path


# -----------------------------------------------------------------------------
# STRATEGIES — paste new ladders here. Each tier is (min_spins_inclusive,
# max_spins_inclusive, bet_multiplier). Tiers must be contiguous and
# non-overlapping starting at spin 1. The tier with the highest
# bet_multiplier is the "top tier" — that's the threshold used for the
# capture/waste metrics.
# -----------------------------------------------------------------------------
STRATEGIES = {
    'reddit_basic_3_50_100': {
        'source': 'example ladder from spec',
        'notes': 'classic 3-tier: safe early, raise mid, bomb late',
        'tiers': [
            (1,   50,  3),
            (51,  90,  50),
            (91,  999, 100),
        ],
    },
}


# -----------------------------------------------------------------------------
# Tier validation
# -----------------------------------------------------------------------------
def validate_strategy(name, strategy):
    """Hard-error if tiers are missing, empty, overlapping, or non-contiguous."""
    tiers = strategy.get('tiers')
    if not tiers:
        raise ValueError(f"strategy {name!r}: 'tiers' is missing or empty")
    if tiers[0][0] != 1:
        raise ValueError(
            f"strategy {name!r}: first tier must start at spin 1, "
            f"got {tiers[0][0]}"
        )
    prev_max = 0
    for i, tier in enumerate(tiers):
        if len(tier) != 3:
            raise ValueError(
                f"strategy {name!r}: tier {i} has {len(tier)} fields, "
                f"expected 3 (min, max, bet)"
            )
        lo, hi, bet = tier
        if lo != prev_max + 1:
            raise ValueError(
                f"strategy {name!r}: tier {i} starts at {lo}, expected "
                f"{prev_max + 1} (tiers must be contiguous)"
            )
        if hi < lo:
            raise ValueError(
                f"strategy {name!r}: tier {i} has max {hi} < min {lo}"
            )
        if bet <= 0:
            raise ValueError(
                f"strategy {name!r}: tier {i} has non-positive bet {bet}"
            )
        prev_max = hi


def top_tier_bet(strategy):
    """Return the highest bet_multiplier across all tiers."""
    return max(bet for (_lo, _hi, bet) in strategy['tiers'])


# -----------------------------------------------------------------------------
# Main
# -----------------------------------------------------------------------------
def main():
    # Validate all strategies up front so bad configs fail fast.
    for name, strategy in STRATEGIES.items():
        validate_strategy(name, strategy)
        print(f"validated strategy: {name:40s} top_tier={top_tier_bet(strategy)}x")
    print(f"\n{len(STRATEGIES)} strategy(ies) validated")


if __name__ == '__main__':
    main()

# -----------------------------------------------------------------------------
# Session-local sa_spins counter + ladder lookup
# -----------------------------------------------------------------------------
def assign_strategy_bets(spins, strategy):
    """
    Walk the per-account spin list and return a parallel list of bet
    multipliers assigned by the strategy's ladder.

    Counter rules (per the spec, session 4.2):
      - local_sa_spins starts at 1 on the first spin of every session
        (session boundary is a hard reset — no pre-CSV history).
      - Counter resets to 1 immediately AFTER any in-session accumulation
        triple (so the NEXT spin starts a fresh gap at 1).
      - Otherwise the counter increments by 1 each spin.

    The bet on spin i is determined by looking up local_sa_spins[i] in
    the ladder tiers. Tiers are (min, max, bet_multiplier), inclusive on
    both ends, contiguous, non-overlapping, starting at 1.
    """
    tiers = strategy['tiers']
    bets = []
    counter = 1
    prev_session_idx = None
    for s in spins:
        sidx = s['session_idx']
        if prev_session_idx is not None and sidx != prev_session_idx:
            counter = 1  # new session = hard reset
        prev_session_idx = sidx

        # Ladder lookup — find the tier whose [lo, hi] contains counter.
        bet = None
        for lo, hi, b in tiers:
            if lo <= counter <= hi:
                bet = b
                break
        if bet is None:
            # counter exceeded the last tier's max — use the last tier's bet
            # as a safety net (validator ensures tiers are contiguous from 1,
            # so this only happens if counter > final tier's max).
            bet = tiers[-1][2]
        bets.append(bet)

        # Advance counter for NEXT spin.
        if s['triple'] == 'accumulation':
            counter = 1
        else:
            counter += 1
    return bets
```

# -----------------------------------------------------------------------------
# Metrics
# -----------------------------------------------------------------------------
def compute_metrics(spins, bets, top_tier):
    """
    Compute capture/waste metrics for a spin stream given a parallel bet
    assignment and a top-tier threshold. Works for both the STRATEGY run
    (bets from the ladder) and the ACTUAL run (bets from real CSV data).

    Returns a dict:
      total_acc_triples : int   — total accumulation triples in the stream
      top_tier_spins    : int   — spins with bet >= top_tier
      captured          : int   — spins at top tier that ARE an acc triple
      wasted            : int   — spins at top tier that are NOT an acc triple
      capture_rate      : float — captured / total_acc_triples   (0..1)
      efficiency        : float — captured / top_tier_spins      (0..1)
      waste_ratio       : float — wasted / captured              (inf if 0)
    """
    assert len(spins) == len(bets), "spins and bets must be same length"
    total_acc = 0
    top_spins = 0
    captured = 0
    for s, b in zip(spins, bets):
        is_acc = s['triple'] == 'accumulation'
        is_top = b >= top_tier
        if is_acc:
            total_acc += 1
        if is_top:
            top_spins += 1
            if is_acc:
                captured += 1
    wasted = top_spins - captured
    capture_rate = (captured / total_acc) if total_acc else 0.0
    efficiency   = (captured / top_spins) if top_spins else 0.0
    waste_ratio  = (wasted / captured) if captured else float('inf')
    return {
        'total_acc_triples': total_acc,
        'top_tier_spins':    top_spins,
        'captured':          captured,
        'wasted':            wasted,
        'capture_rate':      capture_rate,
        'efficiency':        efficiency,
        'waste_ratio':       waste_ratio,
    }
```

# -----------------------------------------------------------------------------
# Verdict
# -----------------------------------------------------------------------------
def format_verdict(ms, ma):
    """
    Deterministic verdict rule per the spec:
      - delta_capture is in percentage points (pp)
      - delta_waste is relative change: (strategy_waste - actual_waste) / actual_waste
        (0 if actual_waste == 0)

    Rules:
      - delta_capture >= +5pp AND delta_waste <= 0       -> 'strategy wins'
      - delta_capture <= -5pp OR  delta_waste >= +0.5    -> 'strategy loses'
      - otherwise                                        -> 'roughly neutral'

    Special case: if the strategy captured nothing (waste_ratio=inf) but
    actual also captured nothing, treat delta_waste as 0. If only one side
    captured nothing, fall through to a verbal verdict.
    """
    delta_cap = ms['capture_rate'] - ma['capture_rate']  # in fraction (0..1)
    delta_cap_pp = delta_cap * 100.0

    # delta_waste: relative change. Handle infinities carefully.
    s_inf = ms['waste_ratio'] == float('inf')
    a_inf = ma['waste_ratio'] == float('inf')
    if s_inf and a_inf:
        delta_waste = 0.0
    elif s_inf and not a_inf:
        delta_waste = float('inf')
    elif not s_inf and a_inf:
        delta_waste = -float('inf')  # strategy went from inf to finite = huge win on waste
    else:
        if ma['waste_ratio'] == 0:
            delta_waste = 0.0 if ms['waste_ratio'] == 0 else float('inf')
        else:
            delta_waste = (ms['waste_ratio'] - ma['waste_ratio']) / ma['waste_ratio']

    if delta_cap_pp >= 5.0 and delta_waste <= 0:
        label = 'strategy WINS'
    elif delta_cap_pp <= -5.0 or delta_waste >= 0.5:
        label = 'strategy LOSES'
    else:
        label = 'roughly neutral'

    delta_waste_str = '+inf' if delta_waste == float('inf') else \
                      '-inf' if delta_waste == -float('inf') else \
                      f"{delta_waste:+.1%}"
    return (
        f"Δcap={delta_cap_pp:+.1f}pp   "
        f"Δwaste={delta_waste_str:>7s}   "
        f"→ {label}"
    )
```

            actual_bets = [s['bet_multiplier'] for s in spins]
            ms = compute_metrics(spins, strat_bets, top)
            ma = compute_metrics(spins, actual_bets, top)

            def fmt(m):
                waste_str = f"{m['waste_ratio']:.1f}:1" if m['captured'] else "  --"
                return (
                    f"captured {m['captured']:4d}/{m['total_acc_triples']:<4d} "
                    f"({m['capture_rate']:5.1%})   "
                    f"eff {m['captured']:4d}/{m['top_tier_spins']:<5d} "
                    f"({m['efficiency']:5.2%})   "
                    f"waste {waste_str:>7s}"
                )

            block = [
                f"\n{account.upper():6s} ({n_spins} spins, "
                f"{ms['total_acc_triples']} acc triples)",
                f"  STRATEGY:  {fmt(ms)}",
                f"  ACTUAL:    {fmt(ma)}",
                f"  VERDICT:   {format_verdict(ms, ma)}",
            ]
            for line in block:
                print(line)
                report_lines.append(line)
```

Note: `report_lines` is collected here and will be written to a file in Task 7. Task 6 does not write the file yet.
